# Folha de Pagamento v2 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Substituir o módulo de folha por um motor de rubricas genérico, automatizado mês a mês, com provisões, conciliação com despesas, holerite PDF e pacote do contador.

**Architecture:** Schema novo `folha_*` (catálogo de rubricas + perfis de cálculo + lançamentos auditáveis); motor de cálculo puro em `src/lib/folha/engine/` (zero Supabase); server actions para workflow de estados; cron dispatcher único diário; saídas em jsPDF/ExcelJS. Spec: `docs/superpowers/specs/2026-06-12-folha-v2-design.md`.

**Tech Stack:** Next.js 14 App Router, TypeScript, Supabase (Postgres + RLS), Vitest, jsPDF, ExcelJS, Vercel Cron.

**Convenções obrigatórias do projeto:** UI via `@/components/ui/*`; `money.format()` de `@/lib/constants`; competência `YYYY-MM`; datas ISO; server actions com `requirePermission(modulo, acao)`; `DEFAULT_SCHOOL_ID`; zero comentários por default.

```typescript
const FERIADOS_NACIONAIS_FIXOS = ["01-01", "04-21", "05-01", "09-07", "10-12", "11-02", "11-15", "11-20", "12-25"];

export function isDiaUtil(dataISO: string, feriadosLocais: string[]): boolean {
  const d = new Date(`${dataISO}T12:00:00Z`);
  const dow = d.getUTCDay();
  if (dow === 0 || dow === 6) return false;
  if (FERIADOS_NACIONAIS_FIXOS.includes(dataISO.slice(5))) return false;
  return !feriadosLocais.includes(dataISO);
}

export function nthDiaUtil(competencia: string, n: number, feriadosLocais: string[]): string {
  const [ano, mes] = competencia.split("-").map(Number);
  let count = 0;
  for (let dia = 1; dia <= 31; dia++) {
    const iso = `${ano}-${String(mes).padStart(2, "0")}-${String(dia).padStart(2, "0")}`;
    const d = new Date(`${iso}T12:00:00Z`);
    if (d.getUTCMonth() + 1 !== mes) break;
    if (isDiaUtil(iso, feriadosLocais)) {
      count++;
      if (count === n) return iso;
    }
  }
  throw new Error(`Sem ${n}º dia útil em ${competencia}`);
}
```

```typescript
import { calcIR, type IrBracket } from "@/lib/payroll/calculators";

export type RedutorIrrf = { limite_isencao: number; limite_reducao: number };

function round2(n: number): number {
  return Math.round(n * 100) / 100;
}

export function calcIrrf2026(args: {
  baseIr: number;
  rendimento: number;
  dependentes: number;
  faixas: IrBracket[];
  redutor: RedutorIrrf | null;
}): number {
  const integral = calcIR(args.baseIr, args.dependentes, args.faixas);
  if (!args.redutor) return integral;
  const { limite_isencao, limite_reducao } = args.redutor;
  if (args.rendimento <= limite_isencao) return 0;
  if (args.rendimento >= limite_reducao) return integral;
  const fator = (limite_reducao - args.rendimento) / (limite_reducao - limite_isencao);
  return round2(Math.max(0, integral - integral * fator));
}
```

```typescript
import { calcINSS, type InssBracket, type IrBracket } from "@/lib/payroll/calculators";
import { calcIrrf2026, type RedutorIrrf } from "./irrf";
import { calcDsrSobre, calcHoraAtividade, calcHoraAula } from "./proventos";
import type {
  ContratoCalculo, LancamentoCalculado, LancamentoManual,
  PerfilRubrica, ResultadoItem, RubricaDef, ValidacaoItem,
} from "./types";

function round2(n: number): number {
  return Math.round(n * 100) / 100;
}

export function calcularItem(args: {
  contrato: ContratoCalculo;
  perfilRubricas: PerfilRubrica[];
  config: { divisor_dsr: number; percentual_hora_atividade: number; semanas_mes: number };
  manuais: LancamentoManual[];
  faixas: { inss: InssBracket[]; ir: IrBracket[] };
  redutor: RedutorIrrf | null;
  rubricasExtras?: RubricaDef[];
}): ResultadoItem {
  const { contrato, config } = args;
  const lancamentos: LancamentoCalculado[] = [];
  const validacoes: ValidacaoItem[] = [];
  const rubricaPorCodigo = new Map<string, RubricaDef>();
  for (const p of args.perfilRubricas) rubricaPorCodigo.set(p.rubrica.codigo, p.rubrica);
  for (const r of args.rubricasExtras ?? []) rubricaPorCodigo.set(r.codigo, r);

  const ordenadas = [...args.perfilRubricas].sort((a, b) => a.ordem_execucao - b.ordem_execucao);
  const encargos = { fgts: 0, inss_patronal: 0, provisao_13: 0, provisao_ferias: 0 };

  const somaPor = (pred: (r: RubricaDef) => boolean) =>
    round2(lancamentos.reduce((acc, l) => {
      const r = rubricaPorCodigo.get(l.rubrica_codigo);
      if (!r || r.tipo !== "provento" || !pred(r)) return acc;
      return acc + l.valor;
    }, 0));

  for (const { rubrica } of ordenadas.filter((p) => p.automatica)) {
    switch (rubrica.metodo_calculo) {
      case "salario_base": {
        if (contrato.salario_base == null) {
          validacoes.push({ nivel: "erro", mensagem: `Contrato sem salário base (${rubrica.codigo})` });
          break;
        }
        lancamentos.push({ rubrica_codigo: rubrica.codigo, referencia: null, valor: round2(contrato.salario_base), origem: "auto" });
        break;
      }
      case "hora_aula": {
        if (contrato.valor_hora_aula == null || contrato.aulas_semanais == null) {
          validacoes.push({ nivel: "erro", mensagem: "Contrato professor sem hora-aula/aulas semanais" });
          break;
        }
        const v = calcHoraAula(contrato.valor_hora_aula, contrato.aulas_semanais, config.semanas_mes);
        lancamentos.push({ rubrica_codigo: rubrica.codigo, referencia: `${contrato.aulas_semanais} aulas`, valor: v, origem: "auto" });
        break;
      }
      case "valor_contratual": {
        const verba = contrato.verbas.find((x) => x.rubrica_codigo === rubrica.codigo);
        if (!verba?.valor) break;
        lancamentos.push({ rubrica_codigo: rubrica.codigo, referencia: null, valor: round2(verba.valor), origem: "auto" });
        break;
      }
      case "dsr": {
        for (const l of [...lancamentos]) {
          const r = rubricaPorCodigo.get(l.rubrica_codigo);
          if (r?.incide_dsr) {
            lancamentos.push({
              rubrica_codigo: rubrica.codigo, referencia: `s/ ${r.nome}`,
              valor: calcDsrSobre(l.valor, config.divisor_dsr), origem: "auto",
            });
          }
        }
        break;
      }
      case "hora_atividade": {
        if (config.percentual_hora_atividade <= 0) break;
        const base = somaPor((r) => r.incide_dsr || r.metodo_calculo === "dsr");
        lancamentos.push({
          rubrica_codigo: rubrica.codigo, referencia: `${config.percentual_hora_atividade}%`,
          valor: calcHoraAtividade(base, config.percentual_hora_atividade), origem: "auto",
        });
        break;
      }
    }
  }

  for (const m of args.manuais) {
    lancamentos.push({
      rubrica_codigo: m.rubrica_codigo, referencia: m.referencia ?? null,
      valor: round2(m.valor), origem: m.origem,
    });
  }

  const base_inss = somaPor((r) => r.incide_inss);
  const base_fgts = somaPor((r) => r.incide_fgts);
  const rendimentoIrrf = somaPor((r) => r.incide_irrf);

  let inssValor = 0;
  for (const { rubrica } of ordenadas.filter((p) => p.automatica)) {
    switch (rubrica.metodo_calculo) {
      case "inss":
        inssValor = calcINSS(base_inss, args.faixas.inss);
        lancamentos.push({ rubrica_codigo: rubrica.codigo, referencia: null, valor: inssValor, origem: "auto" });
        break;
      case "inss_rpa": {
        const teto = args.faixas.inss.reduce((mx, b) => Math.max(mx, b.valor_ate ?? mx), 0);
        const baseRpa = somaPor((r) => r.tipo === "provento");
        inssValor = round2(Math.min(baseRpa, teto || baseRpa) * 0.11);
        lancamentos.push({ rubrica_codigo: rubrica.codigo, referencia: "11%", valor: inssValor, origem: "auto" });
        break;
      }
      case "irrf": {
        const base_irrf = round2(rendimentoIrrf > 0 ? rendimentoIrrf - inssValor
          : somaPor((r) => r.tipo === "provento") - inssValor);
        const v = calcIrrf2026({
          baseIr: base_irrf, rendimento: rendimentoIrrf > 0 ? rendimentoIrrf : base_irrf,
          dependentes: contrato.dependentes_irrf, faixas: args.faixas.ir, redutor: args.redutor,
        });
        lancamentos.push({ rubrica_codigo: rubrica.codigo, referencia: null, valor: v, origem: "auto" });
        break;
      }
      case "percentual_sobre_base": {
        const verba = contrato.verbas.find((x) => x.rubrica_codigo === rubrica.codigo);
        const pct = verba?.percentual;
        if (!pct) break;
        lancamentos.push({
          rubrica_codigo: rubrica.codigo, referencia: `${pct}%`,
          valor: round2(base_inss * (pct / 100)), origem: "auto",
        });
        break;
      }
      case "fgts":
        encargos.fgts = round2(base_fgts * 0.08);
        break;
      case "inss_patronal": {
        const baseTotal = somaPor((r) => r.tipo === "provento");
        encargos.inss_patronal = round2((base_inss > 0 ? base_inss : baseTotal) * 0.2);
        break;
      }
      case "provisao_13":
        encargos.provisao_13 = round2(base_fgts / 12);
        break;
      case "provisao_ferias":
        encargos.provisao_ferias = round2((base_fgts * 4) / 3 / 12);
        break;
    }
  }

  const total_proventos = round2(lancamentos
    .filter((l) => rubricaPorCodigo.get(l.rubrica_codigo)?.tipo === "provento")
    .reduce((a, l) => a + l.valor, 0));
  const total_descontos = round2(lancamentos
    .filter((l) => rubricaPorCodigo.get(l.rubrica_codigo)?.tipo === "desconto")
    .reduce((a, l) => a + l.valor, 0));
  const liquido = round2(total_proventos - total_descontos);
  const base_irrf = round2(Math.max(0, rendimentoIrrf - inssValor));

  if (liquido < 0) validacoes.push({ nivel: "erro", mensagem: "Líquido negativo" });
  if (total_proventos > 0 && base_inss === 0 && rubricaPorCodigo.has("inss"))
    validacoes.push({ nivel: "erro", mensagem: "Provento sem base INSS" });
  if (total_proventos === 0)
    validacoes.push({ nivel: "erro", mensagem: "Contracheque sem proventos" });

  return { lancamentos, total_proventos, total_descontos, liquido, base_inss, base_irrf, base_fgts, encargos, validacoes };
}
```

Nota de implementação: DSR de lançamentos manuais com `incide_dsr` (ex.: HE 50%) só entra se o manual existir antes do recálculo — o serviço (Task 8) sempre recalcula o item inteiro após qualquer edição, então a ordem se resolve lá.

```typescript
export async function gerarRun(companyId: string, competencia: string, geradaPor: string) {
  const supabase = await createServerClient();

  const existente = await supabase.from("folha_runs").select("id")
    .eq("company_id", companyId).eq("competencia", competencia).maybeSingle();
  if (existente.data) return { runId: existente.data.id, criada: false };

  const [{ data: config }, { data: contratos }, { data: escola }] = await Promise.all([
    supabase.from("folha_config").select("*").eq("company_id", companyId).single(),
    supabase.from("folha_contratos")
      .select("*, folha_contratos_rubricas(valor, percentual, ativa, folha_rubricas(codigo))")
      .eq("company_id", companyId).eq("ativo", true),
    supabase.from("companies").select("id").eq("id", companyId).single(),
  ]);
  if (!config || !contratos || !escola) throw new Error("Config/contratos não encontrados");

  const { data: run, error } = await supabase.from("folha_runs")
    .insert({ escola_id: config.escola_id, company_id: companyId, competencia, gerada_por: geradaPor })
    .select("id").single();
  if (error) throw error;

  const recorrentes = await buscarRecorrentesDoMesAnterior(companyId, competencia);

  for (const contrato of contratos) {
    const manuais = recorrentes.get(contrato.id) ?? [];
    await recalcularItemDb(run.id, contrato.id, manuais);
  }
  await recalcularTotaisRun(run.id);
  return { runId: run.id, criada: true };
}

async function buscarRecorrentesDoMesAnterior(companyId: string, competencia: string) {
  const supabase = await createServerClient();
  const [ano, mes] = competencia.split("-").map(Number);
  const anterior = mes === 1 ? `${ano - 1}-12` : `${ano}-${String(mes - 1).padStart(2, "0")}`;
  const { data } = await supabase.from("folha_lancamentos")
    .select(`valor, referencia, recorrente_parcelas, recorrente_parcela_atual,
      folha_rubricas(codigo), folha_itens!inner(contrato_id, folha_runs!inner(company_id, competencia))`)
    .eq("folha_itens.folha_runs.company_id", companyId)
    .eq("folha_itens.folha_runs.competencia", anterior)
    .in("origem", ["manual", "recorrente"])
    .not("recorrente_parcelas", "is", null);
  const porContrato = new Map<string, LancamentoManual[]>();
  for (const l of data ?? []) {
    const atual = (l.recorrente_parcela_atual ?? 0) + 1;
    if (l.recorrente_parcelas != null && atual > l.recorrente_parcelas) continue;
    const arr = porContrato.get(l.folha_itens.contrato_id) ?? [];
    arr.push({
      rubrica_codigo: l.folha_rubricas.codigo, valor: l.valor,
      referencia: `${atual}/${l.recorrente_parcelas}`, origem: "recorrente",
      recorrente_parcelas: l.recorrente_parcelas, recorrente_parcela_atual: atual,
    });
    porContrato.set(l.folha_itens.contrato_id, arr);
  }
  return porContrato;
}

export async function recalcularItemDb(runId: string, contratoId: string, manuaisOverride?: LancamentoManual[]) {
  const supabase = await createServerClient();
  const { data: run } = await supabase.from("folha_runs")
    .select("competencia, company_id, status").eq("id", runId).single();
  if (!run) throw new Error("Run não encontrada");
  if (!["rascunho", "em_revisao"].includes(run.status)) throw new Error("Run travada para edição");

  const [{ data: contrato }, { data: config }, faixas] = await Promise.all([
    supabase.from("folha_contratos")
      .select("*, folha_contratos_rubricas(valor, percentual, ativa, folha_rubricas(codigo))")
      .eq("id", contratoId).single(),
    supabase.from("folha_config").select("*").eq("company_id", run.company_id).single(),
    getFaixasVigentes(run.competencia),
  ]);
  if (!contrato || !config) throw new Error("Contrato/config não encontrados");

  const { data: perfilRubricas } = await supabase.from("folha_perfis_rubricas")
    .select("automatica, ordem_execucao, folha_rubricas(*)")
    .eq("perfil_id", contrato.perfil_calculo_id);
  const { data: todasRubricas } = await supabase.from("folha_rubricas").select("*")
    .eq("escola_id", contrato.escola_id).eq("ativa", true);

  let item = (await supabase.from("folha_itens").select("id")
    .eq("run_id", runId).eq("contrato_id", contratoId).maybeSingle()).data;

  let manuais = manuaisOverride;
  if (!manuais && item) {
    const { data: existentes } = await supabase.from("folha_lancamentos")
      .select("valor, referencia, origem, recorrente_parcelas, recorrente_parcela_atual, folha_rubricas(codigo)")
      .eq("item_id", item.id).in("origem", ["manual", "recorrente"]);
    manuais = (existentes ?? []).map((l) => ({
      rubrica_codigo: l.folha_rubricas.codigo, valor: l.valor,
      referencia: l.referencia ?? undefined, origem: l.origem as "manual" | "recorrente",
      recorrente_parcelas: l.recorrente_parcelas ?? undefined,
      recorrente_parcela_atual: l.recorrente_parcela_atual ?? undefined,
    }));
  }

  const resultado = calcularItem({
    contrato: {
      id: contrato.id, salario_base: contrato.salario_base,
      valor_hora_aula: contrato.valor_hora_aula, aulas_semanais: contrato.aulas_semanais,
      dependentes_irrf: contrato.dependentes_irrf,
      verbas: (contrato.folha_contratos_rubricas ?? [])
        .filter((v: { ativa: boolean }) => v.ativa)
        .map((v: { valor: number | null; percentual: number | null; folha_rubricas: { codigo: string } }) => ({
          rubrica_codigo: v.folha_rubricas.codigo, valor: v.valor, percentual: v.percentual,
        })),
    },
    perfilRubricas: (perfilRubricas ?? []).map((p) => ({
      rubrica: p.folha_rubricas, automatica: p.automatica, ordem_execucao: p.ordem_execucao,
    })),
    config: {
      divisor_dsr: config.divisor_dsr,
      percentual_hora_atividade: Number(config.percentual_hora_atividade),
      semanas_mes: Number(config.semanas_mes),
    },
    manuais: manuais ?? [],
    faixas: { inss: faixas.inss, ir: faixas.ir },
    redutor: faixas.redutor,
    rubricasExtras: todasRubricas ?? [],
  });

  if (!item) {
    item = (await supabase.from("folha_itens")
      .insert({ run_id: runId, contrato_id: contratoId }).select("id").single()).data;
  }
  if (!item) throw new Error("Falha ao criar item");

  await supabase.from("folha_lancamentos").delete().eq("item_id", item.id);
  const rubricaIds = new Map((todasRubricas ?? []).map((r) => [r.codigo, r.id]));
  const linhas = resultado.lancamentos.map((l, i) => {
    const m = (manuais ?? []).find((x) => x.rubrica_codigo === l.rubrica_codigo && x.valor === l.valor);
    return {
      item_id: item!.id, rubrica_id: rubricaIds.get(l.rubrica_codigo),
      referencia: l.referencia, valor: l.valor, origem: l.origem,
      recorrente_parcelas: m?.recorrente_parcelas ?? null,
      recorrente_parcela_atual: m?.recorrente_parcela_atual ?? null,
    };
  }).filter((l) => l.rubrica_id);
  if (linhas.length) await supabase.from("folha_lancamentos").insert(linhas);

  await supabase.from("folha_itens").update({
    total_proventos: resultado.total_proventos, total_descontos: resultado.total_descontos,
    liquido: resultado.liquido, base_inss: resultado.base_inss,
    base_irrf: resultado.base_irrf, base_fgts: resultado.base_fgts,
  }).eq("id", item.id);

  return resultado;
}

export async function recalcularTotaisRun(runId: string) {
  const supabase = await createServerClient();
  const { data: itens } = await supabase.from("folha_itens")
    .select("total_proventos, total_descontos, liquido").eq("run_id", runId).eq("status", "ativo");
  const t = (itens ?? []).reduce((acc, i) => ({
    p: acc.p + Number(i.total_proventos), d: acc.d + Number(i.total_descontos), l: acc.l + Number(i.liquido),
  }), { p: 0, d: 0, l: 0 });
  await supabase.from("folha_runs").update({
    total_proventos: t.p, total_descontos: t.d, total_liquido: t.l,
  }).eq("id", runId);
}
```

```typescript
    .select("*, folha_config:company_id(categoria_despesa_folha, categoria_despesa_encargos, regra_pagamento, feriados_locais, dia_vencimento_gps, dia_vencimento_fgts)")
    .eq("id", runId).single();
  if (!run) throw new Error("Run não encontrada");
  if (!TRANSICOES[run.status]?.includes(destino)) throw new Error(`Transição ${run.status} → ${destino} inválida`);

  if (destino === "aprovada") {
    const pendencias = await validarRun(runId);
    if (pendencias.length) throw new Error(`Pendências bloqueiam aprovação: ${pendencias.join("; ")}`);
    await gerarDespesasDaRun(run);
    await supabase.from("folha_runs").update({
      status: destino, aprovada_por: session.profile.id, aprovada_em: new Date().toISOString(),
    }).eq("id", runId);
  } else if (destino === "paga") {
    await supabase.from("despesas").update({ data_pagamento: new Date().toISOString().slice(0, 10), status: "paga" })
      .eq("folha_run_id", runId).eq("status", "aberta");
    await supabase.from("folha_runs").update({ status: destino, paga_em: new Date().toISOString() }).eq("id", runId);
  } else if (destino === "fechada") {
    await gravarProvisoes(runId, run.competencia);
    await supabase.from("folha_runs").update({
      status: destino, fechada_por: session.profile.id, fechada_em: new Date().toISOString(),
    }).eq("id", runId);
  } else {
    await supabase.from("folha_runs").update({ status: destino }).eq("id", runId);
  }
  revalidatePath("/rh/folha-v2");
}

export async function reabrirRunAction(formData: FormData) {
```

```typescript
async function validarRun(runId: string): Promise<string[]> {
  const supabase = await createServerClient();
  const { data: itens } = await supabase.from("folha_itens")
    .select("liquido, total_proventos, base_inss, folha_contratos(employees(name))")

    if (Number(i.liquido) < 0) pendencias.push(`${nome}: líquido negativo`);

async function gerarDespesasDaRun(run: {
  id: string; escola_id: string; competencia: string; total_liquido: number;
  folha_config: { categoria_despesa_folha: string | null; categoria_despesa_encargos: string | null;
    regra_pagamento: { tipo: string; n: number }; feriados_locais: string[];
    dia_vencimento_gps: number; dia_vencimento_fgts: number };
}) {
  const supabase = await createServerClient();
  const cfg = run.folha_config;
  if (!cfg.categoria_despesa_folha) throw new Error("Configure a categoria de despesa da folha");
  const vencFolha = nthDiaUtil(run.competencia, cfg.regra_pagamento.n, cfg.feriados_locais);
  const [ano, mes] = run.competencia.split("-").map(Number);
  const proxMes = mes === 12 ? `${ano + 1}-01` : `${ano}-${String(mes + 1).padStart(2, "0")}`;

  const { data: itens } = await supabase.from("folha_itens")
    .select("base_inss, base_fgts, total_descontos, liquido").eq("run_id", run.id).eq("status", "ativo");
  const baseInss = (itens ?? []).reduce((a, i) => a + Number(i.base_inss), 0);
  const baseFgts = (itens ?? []).reduce((a, i) => a + Number(i.base_fgts), 0);

  const linhas = [
    { descricao: `Folha ${run.competencia} — líquidos`, valor: run.total_liquido,
      categoria_id: cfg.categoria_despesa_folha, data_vencimento: vencFolha },
    { descricao: `GPS/INSS ${run.competencia}`, valor: Math.round(baseInss * 0.2 * 100) / 100,
      categoria_id: cfg.categoria_despesa_encargos ?? cfg.categoria_despesa_folha,
      data_vencimento: `${proxMes}-${String(cfg.dia_vencimento_gps).padStart(2, "0")}` },
    { descricao: `FGTS ${run.competencia}`, valor: Math.round(baseFgts * 0.08 * 100) / 100,
      categoria_id: cfg.categoria_despesa_encargos ?? cfg.categoria_despesa_folha,
      data_vencimento: `${proxMes}-${String(cfg.dia_vencimento_fgts).padStart(2, "0")}` },
  ].filter((l) => l.valor > 0);

  await supabase.from("despesas").insert(linhas.map((l) => ({
    ...l, escola_id: run.escola_id, competencia: run.competencia,
    status: "aberta", folha_run_id: run.id,
  })));
}

async function gravarProvisoes(runId: string, competencia: string) {
  const supabase = await createServerClient();
  const { data: itens } = await supabase.from("folha_itens")
    .select("contrato_id, base_fgts").eq("run_id", runId).eq("status", "ativo");
  for (const i of itens ?? []) {
    const base = Number(i.base_fgts);
    if (base <= 0) continue;
    const tipos = [
      { tipo: "decimo_terceiro", valor: base / 12 },
      { tipo: "ferias", valor: (base * 4) / 3 / 12 },
      { tipo: "fgts", valor: base * 0.08 },
      { tipo: "inss_patronal", valor: base * 0.2 },
    ];
    for (const t of tipos) {
      const { data: ant } = await supabase.from("folha_provisoes")
        .select("saldo_acumulado").eq("contrato_id", i.contrato_id).eq("tipo", t.tipo)
        .is("baixada_em", null).order("competencia", { ascending: false }).limit(1).maybeSingle();
      await supabase.from("folha_provisoes").insert({
        contrato_id: i.contrato_id, competencia, tipo: t.tipo,
        valor_mes: Math.round(t.valor * 100) / 100,
        saldo_acumulado: Math.round(((ant?.saldo_acumulado ?? 0) + t.valor) * 100) / 100,
      });
    }
  }
}
```

```tsx
"use client";
import { useState } from "react";
import { editarLancamentoAction } from "@/lib/actions/folha";
import { money } from "@/lib/constants";

export function LancamentoRow({ lancamento, editavel }: {
  lancamento: { id: string; valor: number; referencia: string | null; origem: string;
    valor_calculado: number | null; folha_rubricas: { nome: string } };
  editavel: boolean;
}) {
  const [editando, setEditando] = useState(false);
  return (
    <tr className="border-t">
      <td>{lancamento.folha_rubricas.nome}</td>
      <td className="text-xs text-gray-500">{lancamento.referencia}</td>
      <td className="text-xs">
        {lancamento.origem}
        {lancamento.valor_calculado != null && (
          <span title={`calculado: ${money.format(Number(lancamento.valor_calculado))}`}> *</span>
        )}
      </td>
      <td className="text-right">
        {editando ? (
          <form action={editarLancamentoAction} className="inline-flex gap-1">
            <input type="hidden" name="lancamento_id" value={lancamento.id} />
            <input name="valor" type="number" step="0.01" defaultValue={lancamento.valor}
              className="w-28 rounded border px-1 text-right" autoFocus />
            <button type="submit" className="text-blue-600">ok</button>
          </form>
        ) : (
          <button type="button" disabled={!editavel} onClick={() => setEditando(true)}>
            {money.format(Number(lancamento.valor))}
          </button>
        )}
      </td>
    </tr>
  );
}
```

- [ ] **Step 1: Gerador** — seguir o padrão do boletim PDF existente (`grep -r "jsPDF" src/lib` para achar o gerador do boletim e copiar a estrutura de cabeçalho/logo). Conteúdo: empresa (nome, CNPJ), funcionário (nome, CPF, perfil), competência; tabela de lançamentos na `ordem_holerite` com colunas Rubrica | Referência | Proventos | Descontos; rodapé com bases (INSS, IRRF, FGTS), total proventos, total descontos, líquido.

```typescript
import { jsPDF } from "jspdf";
import autoTable from "jspdf-autotable";
import { money } from "@/lib/constants";

export function gerarHoleritePdf(dados: {
  empresa: { nome: string; cnpj: string };
  funcionario: { nome: string; cpf: string };
  competencia: string;
  lancamentos: Array<{ nome: string; referencia: string | null; valor: number; tipo: string }>;
  bases: { inss: number; irrf: number; fgts: number };
  totais: { proventos: number; descontos: number; liquido: number };
}): ArrayBuffer {
  const doc = new jsPDF();
  doc.setFontSize(12);
  doc.text(`${dados.empresa.nome} — CNPJ ${dados.empresa.cnpj}`, 14, 16);
  doc.setFontSize(10);
  doc.text(`Recibo de pagamento — Competência ${dados.competencia}`, 14, 23);
  doc.text(`${dados.funcionario.nome} — CPF ${dados.funcionario.cpf}`, 14, 29);
  autoTable(doc, {
    startY: 35,
    head: [["Rubrica", "Referência", "Proventos", "Descontos"]],
    body: dados.lancamentos.filter((l) => l.tipo !== "informativa").map((l) => [
      l.nome, l.referencia ?? "",
      l.tipo === "provento" ? money.format(l.valor) : "",
      l.tipo === "desconto" ? money.format(l.valor) : "",
    ]),
  });
  const y = (doc as unknown as { lastAutoTable: { finalY: number } }).lastAutoTable.finalY + 8;
  doc.text(`Base INSS: ${money.format(dados.bases.inss)}  Base IRRF: ${money.format(dados.bases.irrf)}  Base FGTS: ${money.format(dados.bases.fgts)}`, 14, y);
  doc.setFontSize(11);
  doc.text(`Proventos: ${money.format(dados.totais.proventos)}  Descontos: ${money.format(dados.totais.descontos)}  Líquido: ${money.format(dados.totais.liquido)}`, 14, y + 8);
  return doc.output("arraybuffer");
}
```
